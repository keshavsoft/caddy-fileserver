Programmatic ESM Library

Directly import into your Node.js automation scripts, provisioning tools, and microservices.

// 1. Import from caddy-fileserver
import { 
  initStaticSite, 
  checkStaticSite, 
  listStaticSites,
  buildStaticRoute 
} from "caddy-fileserver";
// 2. Initialize directory, starter template, Caddy route, reload & probe
const site = await initStaticSite({
  domain: "call.keshavsoft.com",
  root: "/usr/share/call.keshavsoft.com",
  xml: true
});
console.log(`Live at: ${site.domain} (root: ${site.rootPath})`);
// 3. Audit both Ubuntu filesystem & Caddy JSON
const audit = checkStaticSite("call.keshavsoft.com");
console.log(`Folder exists: ${audit.filesystem.exists}, Caddy route: ${audit.caddy.exists}`);
